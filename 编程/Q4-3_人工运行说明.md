```powershell
Set-Location -LiteralPath 'D:\Users\LENOVO\Desktop\数学建模\国赛团队共享工作区\编程'

$preflight = "human_run_logs/q4_3_preflight_$stamp"

$compileCommand = 'python -m compileall -q q4_3_baseline'

$compileExitCode = $LASTEXITCODE

@(
  "command=$compileCommand"
  "exit_code=$compileExitCode"
  "status=$(if ($compileExitCode -eq 0) { 'PASS' } else { 'FAIL' })"
  'output_begin'
  $compileOutput
  'output_end'
) | Set-Content -LiteralPath "$preflight/01_compileall.log" -Encoding utf8

$compileOutput | ForEach-Object { Write-Host $_ }
if ($compileExitCode -ne 0) { throw 'Q4-3 compileall failed' }

python -m unittest `
  tests.test_q4_3_price_contracts `
  tests.test_q4_3_model_contracts `
  tests.test_q4_3_midnight_contracts `
  tests.test_q4_3_toy_solver `
  -v 2>&1 | Tee-Object -FilePath "$preflight/02_tests.log"
if ($LASTEXITCODE -ne 0) { throw 'Q4-3 tests failed' }

python -m q4_3_baseline.cli `
  --config config/q4_3_baseline.json `
  validate-input 2>&1 | Tee-Object -FilePath "$preflight/03_validate_input.json"
if ($LASTEXITCODE -ne 0) { throw 'Q4-3 input identity failed' }

@((Get-Item -LiteralPath config/q4_3_baseline.json)) +
  (Get-ChildItem -LiteralPath q4_3_baseline -Filter '*.py' -File) |
  Get-FileHash -Algorithm SHA256 |
  Format-List | Out-File -Encoding utf8 "$preflight/04_source_hashes.txt"

$inventory = Get-ChildItem -LiteralPath $preflight -File |
  Where-Object Name -ne '05_file_inventory.txt' |
  Select-Object Name, Length, LastWriteTime

$inventory |
  Format-Table -AutoSize |
  Out-String |
  Set-Content -Encoding utf8 -LiteralPath "$preflight/05_file_inventory.txt"

Write-Host "Q4-3 preflight feedback: $preflight"
```

预期：`01_compileall.log` 一定存在并记录 `exit_code=0`、`status=PASS`；成功且无诊断时 `output_begin` 与 `output_end` 之间可以为空。18 个测试应全部 `OK`；`validate-input` 的 `status` 为 `PASS`、`sample_0600_observed_count` 为 1、`formal_boundary_observed_contract` 为 1002、`formal_run_executed` 为 false。

```powershell
$tracks = @(

    --config config/q4_3_baseline.json `

    --allow-formal-run
```
